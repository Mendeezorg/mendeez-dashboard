export default { async fetch(request, env, ctx) { // Handle CORS preflight if (request.method === 'OPTIONS') { return new Response(null, { headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, PUT, OPTIONS', 'Access-Control-Allow-Headers': '*', 'Access-Control-Expose-Headers': 'x-shopify-link', } }) } // Parse params const url = new URL(request.url) const store = url.searchParams.get('store') const token = url.searchParams.get('token') const ep = url.searchParams.get('ep') if (!store || !token || !ep) { return new Response(JSON.stringify({ error: 'Missing params' }), { status: 400, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*', } }) } // Only read body on POST (never on GET — avoids consuming the stream) let shopifyMethod = 'GET' let shopifyBody = null if (request.method === 'POST') { try { const b = await request.json() shopifyMethod = b.shopifyMethod || 'POST' shopifyBody = b.shopifyBody || null } catch (e) {} } // Build Shopify request const shopifyUrl = 'https://' + store + '/admin/api/2024-01' + ep const fetchOpts = { method: shopifyMethod, headers: { 'X-Shopify-Access-Token': token, 'Content-Type': 'application/json', } } if (shopifyBody && (shopifyMethod === 'POST' || shopifyMethod === 'PUT')) { fetchOpts.body = JSON.stringify(shopifyBody) } // Call Shopify const res = await fetch(shopifyUrl, fetchOpts) const body2 = await res.text() // Forward response — CRITICAL: pass Link header as x-shopify-link for pagination return new Response(body2, { status: res.status, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*', 'Access-Control-Expose-Headers': 'x-shopify-link', 'x-shopify-link': res.headers.get('link') || '', } }) } }
